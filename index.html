/* ---------------- PDF 匯出（自選方向 + 防裁切） ---------------- */ function setPdfOrientation(val) { selectedPdfOrientation = val; } function exportReportPDF(filename) { const isPortrait = selectedPdfOrientation === 'portrait'; const marginMm = 8; const pageWidthMm = (isPortrait ? 210 : 297) - marginMm * 2; const contentPx = Math.floor(pageWidthMm * 96 / 25.4) - 4; const reportHtml = generateReportHTML(true); const tempContainer = document.createElement('div'); tempContainer.style.cssText = `position: fixed; left: -10000px; top: 0; width: ${contentPx}px; background: #ffffff;`; tempContainer.innerHTML = reportHtml; document.body.appendChild(tempContainer); const reportEl = tempContainer.firstElementChild; reportEl.style.width = contentPx + 'px'; reportEl.style.maxWidth = contentPx + 'px'; reportEl.style.margin = '0'; reportEl.style.boxSizing = 'border-box'; reportEl.style.overflow = 'hidden'; reportEl.querySelectorAll('table').forEach(t => { t.style.maxWidth = '100%'; }); reportEl.querySelectorAll('td, th').forEach(c => { c.style.overflowWrap = 'anywhere'; }); if (isPortrait) { reportEl.querySelectorAll('td div').forEach(c => { c.style.wordBreak = 'break-all'; }); } const prevScrollX = window.scrollX, prevScrollY = window.scrollY; window.scrollTo(0, 0); const cleanup = () => { if (tempContainer.parentNode) document.body.removeChild(tempContainer); window.scrollTo(prevScrollX, prevScrollY); }; const opt = { margin: [marginMm, marginMm, marginMm, marginMm], filename: filename, image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2, useCORS: true, scrollX: 0, scrollY: 0, windowWidth: document.documentElement.clientWidth }, jsPDF: { unit: 'mm', format: 'a4', orientation: selectedPdfOrientation }, pagebreak: { mode: ['avoid-all', 'css', 'legacy'] } }; if (typeof html2pdf === 'undefined') { cleanup(); printWithOrientation(); return; } html2pdf().set(opt).from(reportEl).save() .then(cleanup) .catch(err => { console.error('PDF 匯出失敗，改用列印：', err); cleanup(); printWithOrientation(); }); } function printWithOrientation() { const styleEl = document.createElement('style'); styleEl.textContent = `@page { size: A4 ${selectedPdfOrientation}; margin: 8mm; }`; document.head.appendChild(styleEl); window.print(); setTimeout(() => styleEl.remove(), 1000); } function downloadWeeklyReportPDF() { try { exportReportPDF(`PM_Weekly_Report_${currentTeam || 'Group'}_${weeklyReportDate}.pdf`); } catch (e) { console.error(e); printWithOrientation(); } } function downloadDailyReportPDF() { try { exportReportPDF(`Daily_Report_${currentTeam || 'Group'}_${dailyReportDate}.pdf`); } catch (e) { console.error(e); printWithOrientation(); } }
